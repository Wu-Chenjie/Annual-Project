cmake_minimum_required(VERSION 3.16)
project(annual_swarm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp geometry_msgs nav_msgs std_msgs trajectory_msgs actuator_msgs tf2 tf2_geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
find_package(gz_sim_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
# Reuse algorithms, never the legacy numerical plant / simulation loop.
set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/../../../next_project/cpp)
add_library(annual_algorithms STATIC ${CORE}/src/controller.cpp ${CORE}/src/smc.cpp
  ${CORE}/src/allocator.cpp ${CORE}/src/occupancy_grid.cpp ${CORE}/src/map_loader.cpp)
target_include_directories(annual_algorithms PUBLIC ${CORE}/include)
foreach(node planner controller)
  add_executable(${node}_node src/${node}_node.cpp)
  ament_target_dependencies(${node}_node rclcpp geometry_msgs nav_msgs std_msgs trajectory_msgs actuator_msgs tf2 tf2_geometry_msgs)
  target_link_libraries(${node}_node annual_algorithms)
  install(TARGETS ${node}_node DESTINATION lib/${PROJECT_NAME})
endforeach()
add_library(annual_motor_system SHARED src/motor_system.cpp)
target_link_libraries(annual_motor_system gz-sim8::gz-sim8 gz-transport13::gz-transport13 gz-plugin2::gz-plugin2)
install(TARGETS annual_motor_system DESTINATION lib)
install(DIRECTORY launch config models worlds DESTINATION share/${PROJECT_NAME})
install(DIRECTORY ${CORE}/../maps DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/audit_fusion_run.py scripts/audit_decentralized_run.py scripts/localization_measurement_node.py scripts/state_estimator_node.py scripts/pointcloud_mapping_node.py scripts/formation_node.py scripts/metrics_node.py scripts/smoke_test.py scripts/planning_node.py scripts/candidate_pool_node.py scripts/dynamic_obstacle_node.py scripts/validate_matrix.py scripts/compare_recovery.py scripts/record_decentralized_demo.py scripts/decentralized_smoke.py scripts/decentralized_agent_node.py scripts/view_executor_node.py scripts/exploration_experiment_node.py scripts/exploration_visualization_node.py scripts/search_node.py scripts/search_executor_node.py scripts/search_smoke.py scripts/compare_search.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/planning_runtime.py scripts/candidate_runtime.py scripts/exploration_palette.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY ${CORE}/../core DESTINATION lib/${PROJECT_NAME}/legacy PATTERN "__pycache__" EXCLUDE)
install(FILES scripts/scene.py scripts/exploration_palette.py DESTINATION share/${PROJECT_NAME}/launch)
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(fusion_test test/test_fusion.py TIMEOUT 180)
  ament_add_pytest_test(exploration_priority_test test/test_exploration_priority.py TIMEOUT 180)
  ament_add_pytest_test(integrated_priority_test test/test_integrated_priority.py TIMEOUT 180)
  ament_add_pytest_test(priority_ros_test test/test_priority_ros.py TIMEOUT 60)
  ament_add_pytest_test(decentralized_test test/test_decentralized.py TIMEOUT 180)
  ament_add_pytest_test(exploration_test test/test_exploration.py TIMEOUT 180)
  ament_add_pytest_test(scene_test test/test_scene.py)
  ament_add_pytest_test(candidate_test test/test_candidate_pool.py TIMEOUT 180)
  ament_add_pytest_test(planning_test test/test_planning_runtime.py TIMEOUT 180)
  add_executable(controller_dispatch_test test/controller_dispatch.cpp)
  target_link_libraries(controller_dispatch_test annual_algorithms)
  add_test(NAME controller_dispatch COMMAND controller_dispatch_test)
endif()
ament_package()
